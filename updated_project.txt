name: Deploy DuckStore

on:
  push:
    branches:
      - main

permissions:
  contents: read
  packages: write

jobs:
  validate:
    name: Validate
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: 22

      - name: Node.js syntax checks
        run: |
          node --check app/backend/server.js
          node --check app/frontend/server.js

      - name: Setup Terraform
        uses: hashicorp/setup-terraform@v3

      - name: Terraform checks
        run: |
          terraform fmt -check
          terraform init -backend=false
          terraform validate
        working-directory: terraform

  build-and-push:
    name: Build & Push Images
    needs: validate
    runs-on: ubuntu-latest
    permissions:
      contents: read
      packages: write
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Log in to GitHub Container Registry
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.GITHUB_TOKEN }}

      - name: Lowercase repository name
        id: repo
        run: echo "name=$(echo "${{ github.repository }}" | tr '[:upper:]' '[:lower:]')" >> "$GITHUB_OUTPUT"

      - name: Build & push duckstore-backend
        uses: docker/build-push-action@v6
        with:
          context: app/backend
          file: app/backend/Dockerfile
          push: true
          tags: |
            ghcr.io/${{ steps.repo.outputs.name }}-backend:${{ github.sha }}
            ghcr.io/${{ steps.repo.outputs.name }}-backend:latest

      - name: Build & push duckstore-frontend
        uses: docker/build-push-action@v6
        with:
          context: app/frontend
          file: app/frontend/Dockerfile
          push: true
          tags: |
            ghcr.io/${{ steps.repo.outputs.name }}-frontend:${{ github.sha }}
            ghcr.io/${{ steps.repo.outputs.name }}-frontend:latest

  deploy:
    name: Deploy to EC2
    needs: build-and-push
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Configure AWS credentials
        uses: aws-actions/configure-aws-credentials@v4
        with:
          aws-access-key-id: ${{ secrets.AWS_ACCESS_KEY_ID }}
          aws-secret-access-key: ${{ secrets.AWS_SECRET_ACCESS_KEY }}
          aws-region: eu-west-1

      - name: Deploy workloads via AWS SSM
        run: |
          set -euo pipefail

          REPO_LOWER=$(echo "${{ github.repository }}" | tr '[:upper:]' '[:lower:]')
          IMAGE_TAG="${{ github.sha }}"

          echo "Resolving EC2 instance with tag Role=k3s-server..."
          INSTANCE_ID=$(aws ec2 describe-instances \
            --filters "Name=tag:Role,Values=k3s-server" "Name=instance-state-name,Values=running" \
            --query "Reservations[0].Instances[0].InstanceId" \
            --output text)

          if [ -z "$INSTANCE_ID" ] || [ "$INSTANCE_ID" = "None" ]; then
            echo "Error: No running EC2 instance found with tag Role=k3s-server"
            exit 1
          fi
          echo "Target EC2 Instance: $INSTANCE_ID"

          cat << 'EOF' > deploy.sh
          set -euo pipefail
          export KUBECONFIG=/etc/rancher/k3s/k3s.yaml

          TMP_DIR=$(mktemp -d /tmp/k8s-deploy-XXXXXX)
          trap 'rm -rf "$TMP_DIR"' EXIT

          echo "Pulling updated k8s manifests into $TMP_DIR..."
          git clone --depth 1 "https://x-access-token:__TOKEN__@github.com/__REPO__.git" "$TMP_DIR" 2>/dev/null || git clone --depth 1 "https://github.com/__REPO__.git" "$TMP_DIR"

          cd "$TMP_DIR/k8s"

          echo "Substituting container images with commit SHA tags..."
          sed -i -E 's|image: .*(-backend:|duckstore-backend:).*|image: ghcr.io/__REPO_LOWER__-backend:__SHA__|g' 10-backend.yaml
          sed -i -E 's|image: .*(-frontend:|duckstore-frontend:).*|image: ghcr.io/__REPO_LOWER__-frontend:__SHA__|g' 20-frontend.yaml

          echo "Applying manifests..."
          sudo kubectl apply -f .

          echo "Verifying backend rollout..."
          sudo kubectl -n backend rollout status deployment/duck-api --timeout=120s

          echo "Verifying frontend rollout..."
          sudo kubectl -n frontend rollout status deployment/duck-web --timeout=120s

          echo "Deployment finished successfully."
          EOF

          sed -i "s|__TOKEN__|${{ secrets.GITHUB_TOKEN }}|g" deploy.sh
          sed -i "s|__REPO__|${{ github.repository }}|g" deploy.sh
          sed -i "s|__REPO_LOWER__|$REPO_LOWER|g" deploy.sh
          sed -i "s|__SHA__|$IMAGE_TAG|g" deploy.sh

          python3 -c '
          import json
          with open("deploy.sh") as f:
              commands = [line.strip() for line in f if line.strip() and not line.strip().startswith("#")]
          with open("ssm-parameters.json", "w") as f:
              json.dump({"commands": commands}, f, indent=2)
          '

          echo "Dispatching deployment command via AWS SSM..."
          COMMAND_ID=$(aws ssm send-command \
            --instance-ids "$INSTANCE_ID" \
            --document-name "AWS-RunShellScript" \
            --comment "Deploy DuckStore commit ${{ github.sha }}" \
            --parameters file://ssm-parameters.json \
            --query "Command.CommandId" \
            --output text)

          echo "SSM Command ID: $COMMAND_ID"

          echo "Waiting for command execution to complete..."
          set +e
          aws ssm wait command-executed \
            --command-id "$COMMAND_ID" \
            --instance-id "$INSTANCE_ID"
          WAIT_EXIT=$?
          set -e

          INVOCATION_JSON=$(aws ssm get-command-invocation \
            --command-id "$COMMAND_ID" \
            --instance-id "$INSTANCE_ID" \
            --output json)

          echo "=== STDOUT ==="
          echo "$INVOCATION_JSON" | python3 -c 'import sys, json; print(json.load(sys.stdin).get("StandardOutputContent", ""))'

          echo "=== STDERR ==="
          echo "$INVOCATION_JSON" | python3 -c 'import sys, json; print(json.load(sys.stdin).get("StandardErrorContent", ""))' >&2

          if [ $WAIT_EXIT -ne 0 ]; then
            STATUS=$(echo "$INVOCATION_JSON" | python3 -c 'import sys, json; print(json.load(sys.stdin).get("Status", "Failed"))')
            echo "Error: Deployment command failed with SSM status: $STATUS"
            exit 1
          fi

          echo "Deployment succeeded!"

==> README.md <==
# K3s Infrastructure & Deployment Pipeline

A production-style decoupled architecture provisioning a single-node K3s Kubernetes cluster on AWS EC2 using Terraform, with containerized microservices deployed via GitHub Actions (`.github/workflows/deploy.yml`) and AWS Systems Manager (SSM).

---

## Architecture

The system decouples infrastructure provisioning from software delivery:

```
[ Developer / Git Push to main ]
                │
                ▼
[ GitHub Actions (.github/workflows/deploy.yml) ]
  ├── 1. validate: Node.js syntax & Terraform validation
  ├── 2. build-and-push: Docker images to GHCR (tagged with Git SHA and latest)
  └── 3. deploy: AWS SSM send-command -> EC2 (dynamically resolves Role=k3s-server)
                                            │
                                            ▼
                           [ AWS EC2 Instance: K3s Server ]
                             ├── Syncs & applies Kubernetes manifests (k8s/)
                             ├── Updates Deployments to Git SHA images
                             └── Verifies rollout status completion
```

### Components

1. **Infrastructure (Terraform in `terraform/`)**
   - **VPC & Networking:** Custom VPC (`10.0.0.0/16`), single public subnet (`10.0.1.0/24`), Internet Gateway, and Route Table.
   - **Compute:** Single EC2 instance (`t3.small`, Ubuntu 24.04 LTS) running K3s (control plane and worker on one node).
   - **Persistence & Lifecycle Protection:** `user_data_replace_on_change = false` with `ignore_changes = [user_data, ami]` to protect persistent local-path volumes (SQLite database) from accidental destruction during script updates or upstream AMI upgrades.
   - **Access & Security:** Static Elastic IP (EIP) mapped to the instance. Ingress Security Group opens ports 80 (HTTP) and 443 (HTTPS) to Traefik. Port 22 (SSH) and 6443 (Kubernetes API) remain closed by default.
   - **IAM & Instance Profile:** Attached with `AmazonSSMManagedInstanceCore` for agent-based management via AWS SSM (no bastion or open SSH ports required). IMDSv2 is strictly enforced.
   - **Hardened K3s Bootstrap (`user_data.sh`):** Installs base packages, AWS CLI, and K3s with default `0600` root-only kubeconfig permissions, preventing unprivileged processes from obtaining cluster-admin rights.

2. **Workloads (Kubernetes manifests in `k8s/`)**
   - **`backend` namespace:** Persistent SQLite volume (`PersistentVolumeClaim` backed by `local-path`), `duck-api` Deployment (single replica, Recreate strategy), and internal `ClusterIP` Service on port 3000.
   - **`frontend` namespace:** Stateless `duck-web` Deployment (2 replicas, RollingUpdate strategy), internal `ClusterIP` Service on port 80.
   - **Resource Management:** Memory requests and limits are strictly enforced to protect host memory. CPU limits are intentionally omitted to avoid Linux CFS quota throttling and latency spikes.
   - **Ingress:** Traefik Ingress routing external HTTP/HTTPS traffic to the frontend Service. Frontend queries backend internally via cluster DNS (`duck-api.backend.svc.cluster.local:3000`).

3. **Continuous Integration & Continuous Delivery (`.github/workflows/deploy.yml`)**
   - **`validate`**: Runs static syntax checks on Node.js application files (`node --check`) and Terraform configuration (`terraform fmt -check`, `terraform validate`).
   - **`build-and-push`**: Compiles Docker container images for both frontend and backend, pushes them to GitHub Container Registry (GHCR), and tags each with `${{ github.sha }}` and `latest`.
   - **`deploy`**: Authenticates to AWS, dynamically looks up the running instance ID by tag `Role=k3s-server`, synchronizes `k8s/*.yaml` manifests, updates image tags, and awaits rollout completion using `aws ssm wait command-executed`.

---

## Prerequisites

- **Terraform:** `>= 1.5.0`
- **AWS CLI:** `v2` configured with administrative permissions.
- **Node.js:** `>= 22.0.0` (for local development and validation).
- **Docker:** `>= 24.0` (for local container builds).
- **AWS SSM Session Manager Plugin:** Installed locally for shell access without SSH.
- **GitHub Repository Secrets (for CI/CD):**
  - `AWS_ACCESS_KEY_ID`: IAM access key with EC2 describe and SSM execution permissions.
  - `AWS_SECRET_ACCESS_KEY`: Corresponding secret access key.
  - `GITHUB_TOKEN`: Provided automatically by GitHub Actions with package write permissions.

## Quickstart

### 1. Provision Infrastructure

```bash
# Clone the repository
git clone <repository-url>
cd duckstore

# Configure variables
cp terraform/terraform.tfvars.example terraform/terraform.tfvars
# Modify terraform/terraform.tfvars if required (region, instance type, etc.)

# Initialize and validate
terraform -chdir=terraform init
terraform -chdir=terraform validate

# Plan and apply
terraform -chdir=terraform plan -out=tfplan
terraform -chdir=terraform apply tfplan
```

Review the outputs for the allocated Elastic IP and store URL:
```bash
terraform -chdir=terraform output
```

### 2. Verify Node Bootstrap

Connect to the instance via AWS SSM:
```bash
aws ssm start-session \
  --target $(terraform -chdir=terraform output -raw instance_id) \
  --region $(terraform -chdir=terraform output -raw region)
```

Check bootstrap execution (commands on the host require `sudo` due to `0600` kubeconfig permissions):
```bash
cat /opt/duckstore/.bootstrap-status
sudo kubectl get nodes -o wide
```

### 3. Deploy Workloads via CI/CD

Push changes to the `main` branch to trigger the pipeline (`.github/workflows/deploy.yml`):
```bash
git add .
git commit -m "feat: infrastructure and deployment updates"
git push origin main
```

Monitor execution under the **Actions** tab in GitHub:
1. **validate:** Ensures Node.js and Terraform syntax standards.
2. **build-and-push:** Builds images and pushes them to GHCR.
3. **deploy:** Syncs manifests, updates deployment image tags, and waits for rollout:
   ```bash
   sudo kubectl -n backend rollout status deployment/duck-api
   sudo kubectl -n frontend rollout status deployment/duck-web
   ```

### 4. Teardown

To decommission all cloud resources and prevent ongoing costs:
```bash
terraform -chdir=terraform destroy -auto-approve
```

BEGIN TRANSACTION;

DROP TABLE IF EXISTS order_items;
DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS customers;
DROP TABLE IF EXISTS product_features;
DROP TABLE IF EXISTS products;

DROP VIEW IF EXISTS v_product_cards;
DROP VIEW IF EXISTS v_order_totals;

CREATE TABLE products (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    slug            TEXT    NOT NULL UNIQUE,
    name            TEXT    NOT NULL,
    tagline         TEXT,
    description     TEXT,
    price_cents     INTEGER NOT NULL CHECK (price_cents >= 0),
    badge           TEXT,
    theme           TEXT    NOT NULL,
    body_color      TEXT    NOT NULL,
    bg_color        TEXT    NOT NULL,
    outline_color   TEXT,
    stock           INTEGER NOT NULL DEFAULT 0 CHECK (stock >= 0),
    is_active       INTEGER NOT NULL DEFAULT 1 CHECK (is_active IN (0, 1)),
    created_at      TEXT    NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE product_features (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    product_id  INTEGER NOT NULL,
    label       TEXT    NOT NULL,
    position    INTEGER NOT NULL DEFAULT 0,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE CASCADE
);

CREATE TABLE customers (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    email         TEXT    NOT NULL UNIQUE,
    full_name     TEXT,
    password_hash TEXT,
    created_at    TEXT    NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE orders (
    id           INTEGER PRIMARY KEY AUTOINCREMENT,
    customer_id  INTEGER,
    status       TEXT NOT NULL DEFAULT 'pending'
                 CHECK (status IN ('pending', 'paid', 'shipped', 'cancelled')),
    placed_at    TEXT NOT NULL DEFAULT (datetime('now')),
    FOREIGN KEY (customer_id) REFERENCES customers(id) ON DELETE SET NULL
);

CREATE TABLE order_items (
    id                INTEGER PRIMARY KEY AUTOINCREMENT,
    order_id          INTEGER NOT NULL,
    product_id        INTEGER NOT NULL,
    quantity          INTEGER NOT NULL CHECK (quantity > 0),
    unit_price_cents  INTEGER NOT NULL CHECK (unit_price_cents >= 0),
    FOREIGN KEY (order_id)   REFERENCES orders(id)   ON DELETE CASCADE,
    FOREIGN KEY (product_id) REFERENCES products(id) ON DELETE RESTRICT,
    UNIQUE (order_id, product_id)
);

CREATE INDEX idx_features_product   ON product_features (product_id, position);
CREATE INDEX idx_orders_customer    ON orders           (customer_id);
CREATE INDEX idx_orders_status      ON orders           (status, placed_at);
CREATE INDEX idx_items_order        ON order_items      (order_id);
CREATE INDEX idx_items_product      ON order_items      (product_id);
CREATE INDEX idx_products_active    ON products         (is_active);

CREATE VIEW v_product_cards AS
SELECT
    p.id,
    p.slug,
    p.name,
    p.tagline,
    p.badge,
    p.theme,
    p.body_color,
    p.bg_color,
    p.outline_color,
    p.price_cents,
    PRINTF('%.2f', p.price_cents / 100.0) AS price_display,
    p.stock,
    CASE WHEN p.stock > 0 THEN 'in_stock' ELSE 'sold_out' END AS availability
FROM products p
WHERE p.is_active = 1;

CREATE VIEW v_order_totals AS
SELECT
    o.id            AS order_id,
    o.status,
    o.placed_at,
    c.email         AS customer_email,
    COUNT(oi.id)                            AS line_count,
    COALESCE(SUM(oi.quantity), 0)           AS total_ducks,
    COALESCE(SUM(oi.quantity * oi.unit_price_cents), 0) AS total_cents
FROM orders o
LEFT JOIN customers   c  ON c.id = o.customer_id
LEFT JOIN order_items oi ON oi.order_id = o.id
GROUP BY o.id, o.status, o.placed_at, c.email;

/**
 * Duck Store — Backend API
 * ---------------------------------------------------------------------------
 * Runs in the `backend` namespace. Owns SQLite database and serves JSON.
 */

const express = require("express");
const fs = require("fs");
const path = require("path");
const { DatabaseSync } = require("node:sqlite");

const PORT = Number(process.env.PORT || 3000);

function seedIfEmpty() {
  const { n } = db
    .prepare("SELECT COUNT(*) AS n FROM sqlite_master WHERE type='table' AND name='products'")
    .get();

  if (n === 0) {
    db.exec(fs.readFileSync(SCHEMA_PATH, "utf8"));
    console.log(`[db] seeded a fresh database at ${DB_PATH}`);
  } else {
    console.log(`[db] existing database found at ${DB_PATH}, leaving it alone`);
  }
}
seedIfEmpty();

  const id = Number(req.params.id);
  if (!Number.isInteger(id)) return res.status(400).json({ error: "id must be an integer" });

  const product = qProduct.get(id);
  if (!product) return res.status(404).json({ error: "duck not found" });

  const card = qCard.get(id);
  res.json({
    ...toCard(card),
    description: product.description,
    features: qFeatures.all(id).map((r) => r.label),
  });
});

app.get("/api/ducks/:id/related", (req, res) => {
  const id = Number(req.params.id);
  if (!Number.isInteger(id)) return res.status(400).json({ error: "id must be an integer" });
  res.json(qRelated.all(id).map(toCard));
});

app.post("/api/orders", (req, res) => {
  const productId = Number(req.body?.productId);
  const quantity = Number(req.body?.quantity ?? 1);

  if (!Number.isInteger(productId)) return res.status(400).json({ error: "productId must be an integer" });
  if (!Number.isInteger(quantity) || quantity < 1) return res.status(400).json({ error: "quantity must be >= 1" });

  const product = qProduct.get(productId);
  if (!product) return res.status(404).json({ error: "duck not found" });

  try {
    db.exec("BEGIN");
    const { lastInsertRowid } = insOrder.run();
    insItem.run(lastInsertRowid, quantity, productId);
    db.exec("COMMIT");

    const total = qOrderTotal.get(lastInsertRowid);
    res.status(201).json({
      orderId: Number(lastInsertRowid),
      product: product.name,
      quantity,
      totalCents: total.total_cents,
      total: (total.total_cents / 100).toFixed(2),
    });
  } catch (err) {
    db.exec("ROLLBACK");
    console.error("[api] order failed:", err.message);
    res.status(500).json({ error: "could not place order" });
  }
});

app.get("/api/stats", (req, res) => {
  res.json(qStats.all());
});

app.use((req, res) => res.status(404).json({ error: "no such endpoint" }));

});

    console.log(`[api] ${signal} received, shutting down`);
    server.close(() => {
      db.close();
      process.exit(0);
    });
  });
}

async function api(pathname, options = {}) {
  const res = await fetch(`${BACKEND_URL}${pathname}`, {
    signal: AbortSignal.timeout(5000),
    headers: { "content-type": "application/json" },
    ...options,
  });

  if (!res.ok) {
    const err = new Error(`backend responded ${res.status} for ${pathname}`);
    err.status = res.status;
    throw err;
  }
  return res.json();
}

        method: 'POST',
        headers: { 'content-type': 'application/json' },
        body: JSON.stringify({ productId: productId, quantity: 1 })
      });
      if (!res.ok) throw new Error('order failed');
      const order = await res.json();
      localStorage.setItem('duckCart', getCart() + 1);
      paintCart();
      alert('Order placed successfully. Order #' + order.orderId + ' — $' + order.total);
    } catch (err) {
      alert('Unable to process order. Please try again.');
    } finally {
      button.disabled = false;
    }
  }

  document.addEventListener('DOMContentLoaded', paintCart);
`;

==> terraform/user_data.sh <==
#!/usr/bin/env bash
# =============================================================================
# K3s Node Bootstrap
# =============================================================================
# Runs once via cloud-init on EC2 first boot.
# Responsible ONLY for:
#   1. System packages & tooling
#   2. Installing K3s single-node cluster
#   3. Ensuring the cluster and control plane reach Ready state
# =============================================================================

set -euo pipefail

exec > >(tee -a /var/log/duckstore-bootstrap.log) 2>&1
echo "=== Bootstrap starting at $(date -Is) ==="

ELASTIC_IP="${elastic_ip}"
K3S_VERSION="${k3s_version}"
DOMAIN_NAME="${domain_name}"
DISABLE_METRICS_SERVER="${disable_metrics_server}"

BOOTSTRAP_DIR=/opt/duckstore
STATUS_FILE="$BOOTSTRAP_DIR/.bootstrap-status"

mkdir -p "$BOOTSTRAP_DIR"
echo "starting" > "$STATUS_FILE"

fail() {
  echo "FAILED at step: $1" | tee "$STATUS_FILE"
  exit 1
}

# =============================================================================
# 1. System Packages
# =============================================================================
echo "--- Installing system packages ---"

wait_for_apt() {
  local waited=0
  while fuser /var/lib/dpkg/lock-frontend >/dev/null 2>&1 \
     || fuser /var/lib/apt/lists/lock >/dev/null 2>&1; do
    if [ "$waited" -ge 300 ]; then
      echo "apt lock still held after 300s, continuing anyway"
      return 0
    fi
    echo "waiting for apt lock... ($waited s)"
    sleep 5
    waited=$((waited + 5))
  done
}

export DEBIAN_FRONTEND=noninteractive
wait_for_apt
apt-get update -y || fail "apt-get update"

wait_for_apt
apt-get install -y ca-certificates curl unzip jq git || fail "apt-get install"

if ! command -v aws >/dev/null 2>&1; then
  curl -fsSL "https://awscli.amazonaws.com/awscli-exe-linux-x86_64.zip" -o /tmp/awscliv2.zip \
    || fail "download awscli"
  unzip -q /tmp/awscliv2.zip -d /tmp || fail "unzip awscli"
  /tmp/aws/install || fail "install awscli"
  rm -rf /tmp/aws /tmp/awscliv2.zip
fi
aws --version

# =============================================================================
# 2. Install K3s
# =============================================================================
echo "--- Installing K3s ---"

INSTALL_ARGS="--tls-san $ELASTIC_IP"
if [ -n "$DOMAIN_NAME" ]; then
  INSTALL_ARGS="$INSTALL_ARGS --tls-san $DOMAIN_NAME"
fi

if [ "$DISABLE_METRICS_SERVER" = "true" ]; then
  INSTALL_ARGS="$INSTALL_ARGS --disable metrics-server"
  echo "metrics-server disabled"
fi

if [ "$K3S_VERSION" = "stable" ]; then
  curl -sfL https://get.k3s.io | INSTALL_K3S_CHANNEL=stable sh -s - $INSTALL_ARGS \
    || fail "k3s install"
else
  curl -sfL https://get.k3s.io | INSTALL_K3S_VERSION="$K3S_VERSION" sh -s - $INSTALL_ARGS \
    || fail "k3s install"
fi

export KUBECONFIG=/etc/rancher/k3s/k3s.yaml

echo 'export KUBECONFIG=/etc/rancher/k3s/k3s.yaml' > /etc/profile.d/k3s.sh
chmod 644 /etc/profile.d/k3s.sh

# =============================================================================
# 3. Wait for Cluster Readiness
# =============================================================================
echo "--- Waiting for K3s readiness ---"

echo "Waiting for kubeconfig..."
for i in $(seq 1 60); do
  [ -f /etc/rancher/k3s/k3s.yaml ] && break
  sleep 2
done
[ -f /etc/rancher/k3s/k3s.yaml ] || fail "kubeconfig timeout"

echo "Waiting for Kubernetes API..."
api_up=0
for i in $(seq 1 60); do
  if kubectl get --raw='/readyz' >/dev/null 2>&1; then
    api_up=1
    echo "API server responsive after $((i * 5))s"
    break
  fi
  sleep 5
done
[ "$api_up" = "1" ] || fail "K3s API timeout"

echo "Waiting for node registration..."
node_seen=0
for i in $(seq 1 60); do
  if [ -n "$(kubectl get nodes -o name 2>/dev/null)" ]; then
    node_seen=1
    echo "Node registered after $((i * 5))s"
    break
  fi
  sleep 5
done
[ "$node_seen" = "1" ] || fail "Node registration timeout"

echo "Waiting for node Ready condition..."
kubectl wait --for=condition=Ready node --all --timeout=300s || fail "Node ready timeout"
kubectl get nodes -o wide

# =============================================================================
# Completion
# =============================================================================
{
  echo "ready"
  echo "finished_at=$(date -Is)"
  echo "elastic_ip=$ELASTIC_IP"
} > "$STATUS_FILE"

echo "=== Bootstrap complete at $(date -Is) ==="
